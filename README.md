 📌 Git команды изученные в процессе подготовки 
```bash
# Установка персональных данных
git config --global user.name "Ваше Имя"        # Ваше имя для коммитов
git config --global user.email "ваш@email.com"  # Ваш email для коммитов
# Создание нового репозитория
git init                                             # Инициализировать новый пустой Git-репозиторий в текущей директории

📥 Получение кода
# Клонирование репозитория
git clone ssh\url                               # Создать локальную копию удаленного репозитория
# Скачать изменения без слияния
git fetch origin                                # Загрузить изменения с origin, но не применять их

🔄 Ежедневный workflow
1. Подготовка к работе
# Переключиться на основную ветку
git checkout main                               # Перейти на ветку 'main'
# Получить последние изменения
git pull origin main                            # Синхронизировать локальную 'main' с удаленной
2. Создание рабочей ветки
# Создать и переключиться на новую ветку
git checkout -b new_branch                      # Создать новую ветку для работы над задачей и перейти в неё
git branch new_branch                           # Создать новую ветку
git checkout name_branch                        # Перейти в ветку
3. Работа с изменениями
# Просмотр изменений
git status                                      # Показать статус файлов (измененные, новые, подготовленные)
git diff                                        # Показать детальные изменения, не добавленные в индекс
# Добавление изменений в коммит
git add file.txt                                # Добавить конкретный файл в индекс
git add .                                       # Добавить все измененные/новые файлы в индекс
# Создание коммита
git commit -m "fix:feature_1 "                  # Зафиксировать изменения с описательным сообщением
git commit --amend --no-edit                    # Вносит изменения в последний коммит, но не создает новый
# Отправка изменений
git push -u origin bugfix/login-validation      # Отправить коммиты в удаленную ветку и установить отслеживание

⏪ История изменения
# Просмотр истории изменений коммитов и файлов
git log                                         # Показать полную историю коммитов (хеш, автор, дата, сообщение)
git log --oneline                               # Показать краткую историю коммитов (первые 7 символов хеша и сообщение)
git log -p file.txt                             # Показать полную историю изменений для конкретного файла
# Кто и когда менял конкретные строки
git blame file.txt                              # Показать автора и коммит для каждой строки в файле
# Поиск по истории коммитов
git log --grep="баг"                            # Найти коммиты, сообщения которых содержат "баг"

⏪ Работа с ошибками
# Отмена локальных изменений
git restore file.txt                            # Отменить изменения в файле до последнего коммита
git restore --staged file.txt                   # Убрать файл из индекса, оставив изменения в рабочем каталоге
# Убрать файлы из индекса
git reset                                       # Удаляет файлы из индекса
# Удаление последнего коммита (осторожно!)
git reset --soft HEAD~1                         # Отменить последний коммит, оставив изменения в индексе
# Создание отменяющего коммита (безопасно)
git revert abc1234                              # Создать новый коммит, отменяющий изменения коммита abc1234

🌿 REBASE MERGE
# Быстрое обновление ветки (перебазирование)
git rebase main                                 # Перенести коммиты текущей ветки поверх 'main'
# Слияние !!! Делается из ветки в которую сливать и указывается ветка, которую мержить
git merge new_branch
# Безопасное слияние (создает коммит слияния)
git merge --no-ff feature/new-test              # Слить ветку 'feature/new-test', всегда создавая коммит слияния

🌿 Управление ветками
# Просмотр всех веток
git branch -av                                  # Показать все локальные и удаленные ветки
# Удаление веток
git branch -d old-branch                        # Локальное удаление ветки (только если слита)
git push origin --delete branch                 # Удаление ветки на удаленном сервере
